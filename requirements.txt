# GUI.
# PyQt5 publishes wheels for Windows, macOS and x86_64 Linux only - there is NO
# aarch64 wheel. On the Raspberry Pi 5 pip therefore tries to build Qt/sip from
# source and fails, so install it from apt instead and create the venv with
# --system-site-packages (start.sh already does this):
#     sudo apt install -y python3-pyqt5
# 5.15.11 is the first release whose wheels are built against Python 3.13 (the
# wheels use the stable cp39-abi3 ABI, so they also install on 3.13/3.14).
PyQt5==5.15.11; platform_machine != "aarch64" and platform_machine != "armv7l"

ultralytics>=8.3.0

# OpenCV.
# The regular opencv-python wheel bundles its own copy of the Qt libraries.
# Loaded next to the apt PyQt5 on the Pi that puts two different Qt5 builds in
# one process and the app aborts with "Cannot mix incompatible Qt library" or a
# failed xcb platform plugin. The headless wheel simply omits the unused
# highgui module (all display goes through PyQt5, this code never calls
# cv2.imshow) and is otherwise identical.
# 4.11.0.86 is the first release with cp313 wheels (Python 3.13 support was
# added in that build); 4.9.0.80 has none and pip would try to compile OpenCV
# from source on Python 3.13.
# NOTE: ultralytics hard-depends on the regular opencv-python and pip does NOT
# treat the headless wheel as satisfying it, so on the Pi pip installs
# opencv-python anyway. start.sh uninstalls it again after the install step -
# if you install manually on the Pi, run: pip uninstall -y opencv-python
opencv-python-headless>=4.11.0.86; platform_machine == "aarch64" or platform_machine == "armv7l"
opencv-python>=4.11.0.86; platform_machine != "aarch64" and platform_machine != "armv7l"
# numpy 1.26.x has no cp313 wheels (Python 3.13 support landed in numpy 2.1.0).
# numpy 2.x is compatible with the code here (no removed scalar aliases such as
# np.int/np.float are used) and with ultralytics >=8.3 and opencv >=4.11.
numpy>=2.1.0
# Pillow 10.x has no cp313 wheels; Python 3.13 support shipped in Pillow 11.0.0.
pillow>=11.0.0
pyserial==3.5
python-dotenv==1.0.1
ncnn>=1.0.20250501
